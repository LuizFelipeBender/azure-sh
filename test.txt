# ---------------------------
# API Gateway (HTTP API)
# ---------------------------
info "\n🌐 Configurando API Gateway..."
API_ID="$(aws_call apigatewayv2 get-apis --query "Items[?Name=='${API_NAME}'].ApiId" --output text || true)"

if [ -z "$API_ID" ] || [ "$API_ID" = "None" ]; then
  warn "Criando API Gateway..."
  API_ID="$(aws_call apigatewayv2 create-api --name "$API_NAME" --protocol-type HTTP --query 'ApiId' --output text)"

  INTEGRATION_ID="$(aws_call apigatewayv2 create-integration \
    --api-id "$API_ID" \
    --integration-type AWS_PROXY \
    --integration-uri "arn:aws:lambda:${REGION}:${ACCOUNT_ID}:function:${FUNCTION_NAME}" \
    --payload-format-version 2.0 \
    --query 'IntegrationId' --output text)"

  aws_call apigatewayv2 create-route --api-id "$API_ID" --route-key 'ANY /{proxy+}' --target "integrations/${INTEGRATION_ID}" >/dev/null
  aws_call apigatewayv2 create-route --api-id "$API_ID" --route-key 'ANY /'        --target "integrations/${INTEGRATION_ID}" >/dev/null

  aws_call apigatewayv2 create-stage --api-id "$API_ID" --stage-name prod --auto-deploy >/dev/null

  aws_call lambda add-permission \
    --function-name "$FUNCTION_NAME" \
    --statement-id apigateway-invoke \
    --action lambda:InvokeFunction \
    --principal apigateway.amazonaws.com \
    --source-arn "arn:aws:execute-api:${REGION}:${ACCOUNT_ID}:${API_ID}/*/*" >/dev/null 2>&1 || true

  ok "API Gateway criado"
else
  ok "API Gateway já existe: $API_ID"
fi

API_URL="https://${API_ID}.execute-api.${REGION}.amazonaws.com/prod"
ok "Base URL: ${API_URL}"

echo -e "\n${GREEN}✨ Deploy concluído! ✨${NC}"
echo -e "${CYAN}Testar:${NC} curl ${API_URL}/health"
echo -e "${CYAN}Logs:${NC}  aws logs tail /aws/lambda/${FUNCTION_NAME} --follow --region ${REGION}"
